function speak(a) { if (!("speechSynthesis" in window)) { status.textContent = "Voice support is not available in this browser."; return; } window.speechSynthesis.cancel(); const utterance = new SpeechSynthesisUtterance(a); const voices = window.speechSynthesis.getVoices(); const femaleVoice = voices.find(v => /female|zira|samantha|karen|susan|heera|veena|priya|google.*female/i.test(v.name) ); if (femaleVoice) { utterance.voice = femaleVoice; } utterance.lang = /[\u0900-\u097F]/.test(a) ? "hi-IN" : "en-IN"; utterance.rate = 0.92; utterance.pitch = 1.12; window.speechSynthesis.speak(utterance); }
